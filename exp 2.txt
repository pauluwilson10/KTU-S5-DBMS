CREATE TABLE DEPENDANT(PID NUMBER(38) NOT NULL,F_NAME VARCHAR(25) NOT NULL,L_NAME VARCHAR(25) NOT NULL,PLACE VARCHAR(25) NOT NULL,EMPNO NUMBER(38) NOT NULL);

CREATE TABLE EMPLOYEE(EMP_NO NUMBER(38) NOT NULL,NAME VARCHAR(25) NOT NULL,JOB VARCHAR(25) NOT NULL,SALARY NUMBER(38) NOT NULL,COMM VARCHAR(25),DEP_NO NUMBER(38),SEX VARCHAR(6),FOREIGN KEY(DEP_NO) REFERENCES DEPARTMENT(DEP_NO) ON DELETE CASCADE);
