CREATE TABLE PURCHASE(ORDER_NO VARCHAR(15) NOT NULL PRIMARY KEY,CUST_NO VARCHAR(15) NOT NULL,PRODUCT_NO VARCHAR(15) NOT NULL,QUANTITY INT NOT NULL,ORDER_DATE DATE,FOREIGN KEY(CUST_NO) REFERENCES CUSTOMER(CUST_NO),FOREIGN KEY(PRODUCT_NO) REFERENCES PRODUCT(PRODUCT_NO) ON DELETE CASCADE);
CREATE TABLE PRODUCT(PRODUCT_NO VARCHAR(15) PRIMARY KEY NOT NULL,DESCRIPTION VARCHAR(15) NOT NULL,COMPANY VARCHAR(25) NOT NULL,PRICE INT NOT NULL);
CREATE TABLE CUSTOMER(CUST_NO VARCHAR(15) PRIMARY KEY NOT NULL,NAME VARCHAR(15) NOT NULL,AGE INT NOT NULL,CITY VARCHAR(15) NOT NULL,PINCODE INT NOT NULL,STATE VARCHAR(15) NOT NULL);
CREATE TABLE SUPPLIER( S_NO VARCHAR(25)  PRIMARY KEY NOT NULL,SNAME VARCHAR(15) NOT NULL,AGE INT NOT NULL,CITY VARCHAR(15) NOT NULL);
